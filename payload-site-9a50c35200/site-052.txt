  function xmlNum(root,name){const n=Number(xmlValue(root,name));return Number.isFinite(n)?n:0}
  function xmlParse(text){const doc=new DOMParser().parseFromString(text,'application/xml');if(xmlElements(doc,'parsererror').length||doc.documentElement?.localName==='parsererror')throw new Error('XML inválido ou incompleto.');return doc}
  function validateNfeXml(text,name){
    const doc=xmlParse(text),inf=xmlFirst(doc,'infNFe');
    if(inf){
      const key=(inf.getAttribute('Id')||'').replace(/^NFe/i,'')||xmlValue(doc,'chNFe'),msgs=[];
      if(!/^\d{44}$/.test(key))msgs.push('Chave de acesso ausente ou com tamanho diferente de 44 dígitos.');
      const totals=xmlFirst(inf,'ICMSTot')||doc,n=xmlNum(totals,'vNF'),items=xmlElements(inf,'det'),sum=items.reduce((a,d)=>a+xmlNum(xmlFirst(d,'prod'),'vProd'),0);
      const fields=[['produtos','vProd','Produtos'],['desconto','vDesc','Descontos'],['frete','vFrete','Frete'],['seguro','vSeg','Seguro'],['outrasDespesas','vOutro','Outras despesas'],['ipi','vIPI','IPI'],['icmsSt','vICMSST','ICMS-ST'],['ii','vII','Imposto de Importação'],['ipiDevolvido','vIPIDevol','IPI devolvido'],['icmsDesonerado','vICMSDeson','ICMS desonerado'],['icms','vICMS','ICMS destacado'],['pis','vPIS','PIS'],['cofins','vCOFINS','Cofins'],['fcp','vFCP','FCP'],['fcpSt','vFCPST','FCP-ST'],['tributosAproximados','vTotTrib','Tributos aproximados (informativo)']];
      const components=Object.fromEntries(fields.map(([key,tag])=>{const el=xmlFirst(totals,tag);return[key,el?Number(el.textContent.trim())||0:null]}));
      if(!items.length)msgs.push('Nenhum item de produto foi encontrado.');
      if(n&&sum&&Math.abs(n-sum)>0.05)msgs.push(`Soma dos produtos (${money(sum)}) difere do total da NF (${money(n)}); frete, descontos e outros valores podem explicar parte da diferença.`);
      const emit=xmlFirst(inf,'emit')||xmlFirst(doc,'emit'),dest=xmlFirst(inf,'dest')||xmlFirst(doc,'dest');
      if(!xmlValue(emit,'CNPJ')&&!xmlValue(emit,'CPF'))msgs.push('Emitente sem CNPJ/CPF localizado.');
      if(!xmlValue(dest,'CNPJ')&&!xmlValue(dest,'CPF')&&!xmlValue(dest,'idEstrangeiro'))msgs.push('Destinatário sem identificador localizado.');
      const cfops=[...new Set(items.map(d=>xmlValue(xmlFirst(d,'prod'),'CFOP')).filter(Boolean))],ncmMissing=items.filter(d=>!xmlValue(xmlFirst(d,'prod'),'NCM')).length;
      if(!cfops.length)msgs.push('CFOP não localizado nos itens.');if(ncmMissing)msgs.push(`${ncmMissing} item(ns) sem NCM localizado.`);
      const ide=xmlFirst(inf,'ide'),prot=xmlFirst(doc,'infProt'),cStat=xmlValue(prot,'cStat'),mod=xmlValue(ide,'mod'),emissao=xmlValue(ide,'dhEmi')||xmlValue(ide,'dEmi'),competencia=(emissao.match(/^(\d{4})-(\d{2})/)||[]).slice(1).join('-');
      // A leitura por CFOP é uma estimativa de conciliação: o Livro P9 agrupa valores fiscais e não identifica a chave individual da NF-e.
      const cfopRows={};for(const item of items){const prod=xmlFirst(item,'prod'),cfop=xmlValue(prod,'CFOP');if(!/^\d{4}$/.test(cfop))continue;const first=Number(cfop[0]),direction=[1,2,3].includes(first)?'entrada':[5,6,7].includes(first)?'saida':'indeterminada';if(direction==='indeterminada')continue;const row=cfopRows[`${direction}|${cfop}`]||(cfopRows[`${direction}|${cfop}`]={direction,cfop,produtos:0,desconto:0,frete:0,seguro:0,outrasDespesas:0,ipi:0,icmsSt:0,baseIcms:0,icms:0,notas:0});row.produtos+=xmlNum(prod,'vProd');row.desconto+=xmlNum(prod,'vDesc');row.frete+=xmlNum(prod,'vFrete');row.seguro+=xmlNum(prod,'vSeg');row.outrasDespesas+=xmlNum(prod,'vOutro');row.ipi+=xmlNum(xmlFirst(item,'IPI'),'vIPI');row.icmsSt+=xmlNum(xmlFirst(item,'ICMS'),'vICMSST');row.baseIcms+=xmlNum(xmlFirst(item,'ICMS'),'vBC');row.icms+=xmlNum(xmlFirst(item,'ICMS'),'vICMS');row.notas++;}
      const protocolStatus=['100','150'].includes(cStat)?`Autorizada no XML (cStat ${cStat})`:cStat?`Protocolo no XML com cStat ${cStat}`:'Situação não consta no protocolo deste arquivo';
      return {kind:'nota',name,status:protocolStatus,key,nfe:xmlValue(ide,'nNF'),serie:xmlValue(ide,'serie'),emissao,competencia,emit:xmlValue(emit,'xNome'),dest:xmlValue(dest,'xNome'),emitId:(xmlValue(emit,'CNPJ')||xmlValue(emit,'CPF')).replace(/\D/g,''),destId:(xmlValue(dest,'CNPJ')||xmlValue(dest,'CPF')).replace(/\D/g,''),total:n,items:items.length,cfops:cfops.join(', '),cfopRows:Object.values(cfopRows),components,financialFields:fields.map(([key,,label])=>({key,label})),cStat,model:mod==='65'?'NFC-e':mod==='55'?'NF-e':mod?`Modelo ${mod}`:'NF-e',messages:msgs.length?msgs:['Estrutura básica localizada. A autorização e a situação atual ainda precisam ser confirmadas na SEFAZ.']};
    }
    const eventInfo=xmlFirst(doc,'infEvento'),eventSummary=xmlFirst(doc,'resEvento');
    if(eventInfo||eventSummary){
      const event=eventInfo||eventSummary,ret=xmlFirst(doc,'retEvento'),retInfo=xmlFirst(ret,'infEvento'),eventCode=xmlValue(event,'tpEvento'),key=xmlValue(event,'chNFe'),cStat=xmlValue(retInfo,'cStat')||xml
