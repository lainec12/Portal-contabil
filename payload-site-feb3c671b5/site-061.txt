  // Busca notas incluídas na soma cujo vNF explique a diferença positiva XML − PDF.
  function findDifferenceNoteCandidates(documents,difference){
    const target=Math.round(Number(difference)*100);
    if(!Number.isFinite(target)||target<=0)return{exact:[],combination:[],near:[]};
    const eligible=documents.filter(d=>d.included&&Number(d.note.total)>0)
      .map(d=>({document:d,cents:Math.round(Number(d.note.total)*100)}))
      .filter(d=>d.cents>0&&d.cents<=target+1);
    const exact=eligible.filter(d=>Math.abs(d.cents-target)<=1).map(d=>d.document);
    const near=[...eligible].sort((a,b)=>Math.abs(a.cents-target)-Math.abs(b.cents-target))
      .slice(0,5).map(d=>({document:d.document,delta:(d.cents-target)/100}));
    if(exact.length)return{exact,combination:[],near};
    // Soma de subconjuntos em centavos; limite de estados evita travar lotes grandes no celular.
    const states=new Map([[0,[]]]);let operations=0,combination=[];
    for(let i=0;i<eligible.length&&operations<750000;i++){
      const entries=[...states.entries()];
      for(const [sum,path] of entries){
        operations++;
        const next=sum+eligible[i].cents;
        if(next>target+1)continue;
        if(Math.abs(next-target)<=1){combination=[...path,i];break;}
        if(!states.has(next)&&states.size<18000)states.set(next,[...path,i]);
      }
      if(combination.length)break;
    }
    return{exact,combination:combination.map(i=>eligible[i].document),near};
  }
